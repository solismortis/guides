sudo apt install make

sudo apt install cmake

sudo apt install build-essential

Clone Acados into home directory:
git clone https://github.com/acados/acados.git
cd acados
git submodule update --recursive --init

mkdir build
cd build
cmake .. -DCMAKE_C_FLAGS="-Wno-error=incompatible-pointer-types" -DACADOS_WITH_QPOASES=ON
make clean
make install

Since Acados does not like venv:
sudo apt install python3-virtualenv

Create project folder somewhere, get inside and activate the env:
virtualenv env --python=/usr/bin/python3

Add these vars into "activate" (to the bottom of the script):
export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:"$HOME/acados/lib"
export ACADOS_SOURCE_DIR="$HOME/acados"

cd env/bin
source activate
cd ../..
pip install -e ~/acados/interfaces/acados_template

Testing:
python $HOME/acados/examples/acados_python/getting_started/minimal_example_ocp.py
